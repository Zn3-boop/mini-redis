    char buf[READ_CHUNK];
    ssize_t n = read(c->fd, buf, sizeof buf);
    if (n > 0) {
        if (resp_append(&c->rb, buf, (size_t)n) < 0) { c->dead = 1; return; }
        for (;;) {
            RespCmd cmd;
            int r = resp_parse(&c->rb, &cmd);
            if (r == RESP_NEED_MORE) return;          /* 半包：等下一次事件 */
            if (r == RESP_PROTOCOL_ERROR) { c->dead = 1; return; }

                resp_consume(&c->rb, cmd.consumed);
                continue;
            }
            g_commands++;
            RespOut out; resp_out_init(&out);
            dispatch(&cmd, &out);                     /* ← 命令分发 */
            if (write_all(c->fd, out.buf, out.len) < 0) { c->dead = 1; ... return; }
            resp_out_free(&out);
            free(cmd.argv); free(cmd.argv_len);
            resp_consume(&c->rb, cmd.consumed);
        }
    }
    if (n == 0) { c->dead = 1; return; }              /* 对端关闭 */
    if (errno == EINTR) return;
    if (errno == EAGAIN || errno == EWOULDBLOCK) return; /* 排空完成，连接保留 */
    c->dead = 1;                                      /* 硬错误 */
}

### Q2【进阶】为什么单线程 + epoll？
**考察**：事件驱动模型理解。得分点：① 单线程消除共享状态（无锁、无内存模型问题）；② 省的是上下文切换与 syscall 数，不是 CPU；③ 单命令纯内存 O(N) 不阻塞循环；④ 代价=慢命令卡全局。

**代码索引**——整条事件循环就是单线程：main 里唯一一个 `while` + `epoll_wait`：

### Q3【进阶】事件循环里读/写/定时事件各在什么时机处理？
**考察**：是否理解事件循环的"分派"本质。得分点：**读=epoll_wait 返回后立即处理；写=同步 write_all（EAGAIN 时 poll EPOLLOUT）；定时=本项目没有独立定时器，过期靠命令触发时的惰性检查**。

**代码索引**——写事件（同步阻塞写）：
```c
/* src/server.c:70-86 write_all()  EAGAIN 时原地 poll 等可写 */
static int write_all(int fd, const char *buf, size_t len) {
    size_t off = 0;
    while (off < len) {
        ssize_t n = write(fd, buf + off, len - off);
        if (n > 0) { off += (size_t)n; continue; }
        if (n < 0 && errno == EINTR) continue;
        if (n < 0 && (errno == EAGAIN || errno == EWOULDBLOCK)) {
            struct pollfd p = { .fd = fd, .events = POLLOUT };
            if (poll(&p, 1, -1) <= 0) return -1;
            continue;
        }
        return -1;
    }
    return 0;
}
```

### Q6【进阶】为什么 djb2 + 链地址法？和真 Redis 的取舍？
**考察**：⚠️ 此处常考错误记忆——**真 Redis 不是开放寻址，是链地址 + 渐进式 rehash**。说"真 Redis 用开放寻址"= 基础不牢。得分点：链地址实现简单、删除方便、对坏哈希容忍度高；开放寻址缓存友好但删除要墓碑、负载上限严格。

**代码索引**：
```c
/* src/dict.c:7-17  djb2 + DictNode 链 */
static unsigned long djb2(const char *s) {
    unsigned long h = 5381;
    int c;
    while ((c = (unsigned char)*s++) != 0) h = ((h << 5) + h) + (unsigned)c;
    return h;
}
/* DictNode.next = 链地址法的"链" */
typedef struct DictNode {
    char *key;
    DbValue val;
    struct DictNode *next;   /* ← 冲突时挂链表 */
} DictNode;
```

### Q7【挑战】负载因子 0.75 怎么定？一次性 rehash 的代价？
**考察**：诚实度探测器。得分点：承认一次性 rehash 会暂停服务（新桶分配+全量重哈希），教学级规模下毫秒级可接受；真 Redis 渐进式是"每操作搬一个桶"，需要新旧两表共查。

**代码索引**：
```c
/* src/dict.c:21-39  一次性翻倍重建（非渐进） */
static int dict_rehash(Dict *d) {
    size_t nb = d->nbuckets * 2;
    DictNode **nbk = calloc(nb, sizeof(*nbk));
    if (!nbk) return -1;
    for (size_t i = 0; i < d->nbuckets; i++) {
        DictNode *n = d->buckets[i];           /* 遍历所有旧桶 */
        while (n) {
            DictNode *next = n->next;
            size_t idx = djb2(n->key) % nb;    /* 重算哈希 */
            n->next = nbk[idx]; nbk[idx] = n; n = next;
        }
    }
    free(d->buckets); d->buckets = nbk; d->nbuckets = nb;
    return 0;
}
/* src/dict.c:76  负载因子 0.75 触发 */
if (d->count > d->nbuckets * 3 / 4) dict_rehash(d);
```

### Q10【进阶】now_ms 为什么用 CLOCK_REALTIME 不用 CLOCK_MONOTONIC？
**考察**：墙钟 vs 单调钟。得分点：EXPIRE 的过期点必须是**可持久化的绝对时间戳**（未来 AOF 回放要用 PEXPIREAT 重建）；MONOTONIC 进程重启归零，跨重启 TTL 语义崩。代价：系统改时间会让 key 提前/延后过期，Redis 接受（持久化语义优先）。

**代码索引**：
```c
/* src/server.c:97-101 */
static long long now_ms(void) {
    struct timespec ts;
    clock_gettime(CLOCK_REALTIME, &ts);   /* ← 墙钟，不是 MONOTONIC */
    return (long long)ts.tv_sec * 1000 + ts.tv_nsec / 1000000;
}
/* 过期点在 EXPIRE 里 = now + seconds*1000，存到 g_exp（server.c:202） */
```

    for (DictNode *n = d->buckets[idx]; n; n = n->next) {
        if (strcmp(n->key, key) == 0) {
            if (out) { out->data = n->val.data; out->len = n->val.len; }
            return 0;
        }
    }
    return -1;
}
/* 调用例：server.c:192  EXPIRE 分支判断 key 是否存在 */

## 六、3 个示范答案（背熟）

### 示范答案 A（背诵用）：为什么单线程 epoll 能支撑高并发？瓶颈在哪？
**主干**：对比的是阻塞IO多线程模型——每线程一个连接：C10K 时线程数上万，每次上下文切换和每线程栈内存（默认 8MB）直接把机器拖垮。而事件驱动是**"一个循环管所有连接"**：epoll 告诉我哪个 fd 就绪，我按顺序处理，就绪之间几乎不付额外的调度成本。所以单线程省的是**三条**：上下文切换、线程栈内存、共享数据的锁。真正的瓶颈在于：① 有且只有一个 CPU 核在干活（单核上限）；② 任何一个命令如果 O(N)，会卡住所有其他连接——所以 KEYS * 这种命令必须避免。

**代码证据**（顺势背）：
```c
/* server.c:318 主循环——单线程的全部逻辑都在这个 while 里 */
while (!g_stop) {
    int nfds = epoll_wait(epfd, events, MAX_EVENTS, 1000);   /* 就绪等待 */
    ...
    Conn *c = events[i].data.ptr;
    if (evs & (EPOLLIN | EPOLLRDHUP | EPOLLHUP | EPOLLERR)) handle_conn(epfd, c);
}
```
**加分**：诚实补充"我们为了教学性把写改成了同步 write_all，大回包时会阻塞循环——真 Redis 用输出缓冲，这是我们明确留的差距。"

### 示范答案 B（背诵用）：半包/粘包怎么解决？你的增量解析是怎么设计的？
**主干**：核心是一个**可恢复的状态机**——我不会假设"一次 read 就是一个完整命令"。网络包可能在一行中间断开，也可能一包里塞三个命令，所以：
1）**累积缓冲**：Read 到的数据先 `resp_append` 进 `RespBuf`（resp.c:23-28），不管它完不完整；
2）**三态解析**：`resp_parse` 返回 `RESP_OK / RESP_NEED_MORE / RESP_PROTOCOL_ERROR`（resp.h:21）。NEED_MORE = "还没凑齐，缓冲别动，等下一波事件继续"——天然解决半包；
3）**粘包**：handle_conn 在解析成功后**循环**解析直到 NEED_MORE，一包多命令逐条执行（server.c:255-281）；
4）**边界防护**：argc ≤ 1024、单块 ≤ 64MB（resp.c:9-10），超了 `PROTOCOL_ERROR` 断开，防止脏数据拖垮内存。

**代码证据**：
```c
/* resp.c:157-159 半包场景——块长度行不完整，缓冲原样保留 */
if (scan_crlf(rb, pos, &bl) != 0) {
    free(se);
    return RESP_NEED_MORE;
}
```
**加分**：追问"缓冲区被脏数据堆满怎么办" → "限定最长命令 64MB；超出立刻 ERR 并断开连接，避免攻击者用超长 bulk 持续打内存。这是从 Redis 请求大小限制学的。"

### 示范答案 C（背诵用）：TTL/EXPIRE 的实现和真 Redis 的语义差异？
**主干**：分两个层面讲：
1）**语义层**（对齐真 Redis）：`EXPIRE key seconds` 设置绝对过期点（server.c:202 `now + seconds*1000`）；TTL 返回 -2（不存在）/ -1（没设置过期）/ 剩余秒数——**剩余秒数是四舍五入**不是向下取整，`(rem+500)/1000`（server.c:228）和真 Redis 的 `ttlGenericCommand` 行为一致；`EXPIRE k 0` = 立即删除（server.c:196-201）；`SET` 覆盖旧 key 会顺带清掉过期（server.c:147）；`EXPIRE` 对不存在的 key 返回 0（server.c:192-195）；
2）**存储层**（我们的实现）：**数据表 g_db 和过期表 g_exp 分离**（server.c:31-38），过期只记 key + 绝对时刻的十进制字符串；`expire_if_needed` 在 GET/EXPIRE/TTL 三个命令入口前调用（server.c:119-128）——惰性删除：访问时才发现过期并删除。

**差异点（诚实说出）**：真 Redis 除了惰性删除，还有**每 100ms 的主动抽样删除**（过期 key 较多时回收），我们只做了惰性——追问补充："生产场景会在事件循环里加周期性任务（每 100ms 抽样 100 个 key 主动删），这是文档里明确列出的下一步。"

**代码证据**：
```c
/* server.c:119-128 */
static int expire_if_needed(const char *key, long long now) {
    DbValue v;
    if (dict_get(&g_exp, key, &v) != 0) return 0;   /* 无过期设置 */
    long long at = strtoll(v.data, NULL, 10);
    if (at > now) return 0;                          /* 还没到点 */
    dict_del(&g_db, key);                            /* 两表成对删除 */
    dict_del(&g_exp, key);
    g_expired++;
    return 1;
}
```
